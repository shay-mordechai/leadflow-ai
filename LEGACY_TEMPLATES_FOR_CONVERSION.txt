                const timeStr = `${h < 10 ? '0'+h : h}:00`;
                const fullKey = `${dateStr} ${timeStr}`;
                const isActive = activeSlots.has(fullKey);

                const slot = document.createElement('div');
                slot.className = `slot border rounded-lg py-2.5 text-center font-bold text-xs cursor-pointer ${isActive ? 'active' : 'inactive'}`;
                slot.innerText = timeStr;
                slot.onclick = () => toggleSlot(slot, dateStr, timeStr);
                grid.appendChild(slot);
            }
            wrapper.appendChild(grid);
            container.appendChild(wrapper);
        }

        async function toggleSlot(el, date, time) {
            el.classList.toggle('active');
            el.classList.toggle('inactive');

            try {
                await fetch('/api/v1/schedule/toggle', {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({ date_str: date, time_str: time })
                });
            } catch(e) {
                // Revert on error
                el.classList.toggle('active');
                el.classList.toggle('inactive');
                alert("שגיאת תקשורת");
            }
        }

        async function loadSettings() {

        document.getElementById('settingsForm').onsubmit = async (e) => {
            e.preventDefault();
            const btn = e.target.querySelector('button');
            const originalText = btn.innerText;
            btn.innerHTML = '<i class="fas fa-circle-notch fa-spin"></i> שומר...';
            btn.disabled = true;

            const formData = new FormData(e.target);
            const payload = Object.fromEntries(formData.entries());

            try {
                const res = await fetch('/api/v1/settings/', {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify(payload)
                });
                
                if (res.ok) {
                    alert('✅ המוח עודכן בהצלחה!');
                    window.location.href = '/dashboard';
                } else {
                    alert('שגיאה בשמירה');
                }
            } catch (e) {
                alert('שגיאת תקשורת');
            } finally {
                btn.innerHTML = originalText;
                btn.disabled = false;
            }
        };

        loadSettings();

    <script>
    document.getElementById('loginForm').onsubmit = async (e) => {
        e.preventDefault();
        const btn = e.target.querySelector('button');
        const originalText = btn.innerHTML;
        btn.innerHTML = 'מתחבר...'; 
        btn.disabled = true;

        const formData = new FormData(e.target);
        const emailValue = formData.get('phone'); 
        const rememberMe = formData.get('remember_me') === 'on'; // Check if user checked the box

        const data = {
            email: emailValue,
            password: formData.get('password'),
            // We can send remember_me to server if we implement Refresh Tokens later
            remember_me: rememberMe 
        };

        try {
            console.log('Sending login request...', data);
            const response = await fetch('/api/v1/auth/login', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(data)
            });

            const result = await response.json();

            if (response.ok) {
                console.log('Login successful');

                if (result.access_token) {
                    // 1. Save to LocalStorage (Optional, mostly for API calls)
                    localStorage.setItem('token', result.access_token);
                    
                    // 2. COOKIE LOGIC (Long expiry vs Short expiry)
                    const now = new Date();
                    let expiryTime;

                    if (rememberMe) {
                        // 30 Days if "Remember Me" is checked
                        expiryTime = 30 * 24 * 60 * 60 * 1000;
                    } else {
                        // 24 Hours if NOT checked
                        expiryTime = 24 * 60 * 60 * 1000;
                    }

                    now.setTime(now.getTime() + expiryTime);
                    
                    // Set the cookie with the calculated expiry
                    document.cookie = `access_token=${result.access_token}; expires=${now.toUTCString()}; path=/; SameSite=Lax; Secure`;
                    
                    console.log(`Cookie set. Remember Me: ${rememberMe}. Expires in: ${expiryTime/1000/60/60} hours.`);
                }
                
                window.location.href = '/dashboard';
            } else {
                // If location check failed (401), this block will run
                console.error('Login failed:', result);
                const errorMsg = result.detail?.[0]?.msg || result.detail || 'פרטי התחברות שגויים';
                alert('שגיאה: ' + errorMsg);
                btn.innerHTML = originalText;
                btn.disabled = false;
            }
        } catch (error) {
            console.error('Network Error:', error);
            alert('שגיאת תקשורת. נא לבדוק את החיבור.');
            btn.innerHTML = originalText;
            btn.disabled = false;
        }
    };
</script>
